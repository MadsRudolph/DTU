\section{Exercise 4 – Determine \(V_{B1}\) for \(V_O=1.8\,\mathrm{V}\)}

\subsection*{Given}
\[
\mu_pC_{ox}=9.35\times10^{-6}~\text{A/V}^2,\quad
(W/L)=26.714,\quad
V_T=-0.6~\text{V},\quad
I_D=250~\mu\text{A},\quad
V_{DD}=3.3~\text{V}.
\]

\subsection*{Method}
For a PMOS transistor in saturation:
\begin{equation}
I_D=\tfrac{1}{2}\mu_pC_{ox}\tfrac{W}{L}(V_{SG}-|V_T|)^2,
\label{eq:4.1}
\end{equation}
and
\begin{equation}
V_{\text{eff}}=V_{SG}-|V_T|=\sqrt{\tfrac{2I_D}{\mu_pC_{ox}(W/L)}}.
\label{eq:4.2}
\end{equation}

\subsection*{Calculation}
\[
V_{\text{eff}}=\sqrt{\tfrac{2(250\times10^{-6})}
{(9.35\times10^{-6})(26.714)}}=1.414~\text{V},\quad
V_{SG}=V_{\text{eff}}+|V_T|=2.014~\text{V}.
\]
Since \(V_{SG}=V_{DD}-V_{B1}\):
\[
\boxed{V_{B1}=V_{DD}-V_{SG}=3.3-2.014=1.29~\text{V}.}
\]

\subsection*{Verification}
At \(V_O=1.8~\text{V}\):
\[
V_{SD}=3.3-1.8=1.5~\text{V}>V_{\text{eff}}=1.41~\text{V},
\]
\(M_2\) remains in saturation.
