\section{Exercise 5 – Input capacitance using KCL (Figure~1.1)}

\subsection*{Small-signal model}
The small-signal circuit is shown in Figure~\ref{fig:ss_ex5}. 
$M_1$ is modeled by its transconductance $g_{m1}$, and the PMOS load $M_2$ 
by its small-signal resistance $r_{ds2}$. 
Since $\lambda_2=0.1~\mathrm{V^{-1}}$, channel-length modulation in $M_2$ 
cannot be neglected, so $r_{ds2}$ is included. 
$C_1$ and $C_2$ represent the gate–source and gate–drain capacitances.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\linewidth]{images/small_signal_model_ex5.png}
    \caption{Small-signal circuit.}
    \label{fig:ss_ex5}
\end{figure}

\subsection*{KCL analysis}
At $v_o$:
\[
\frac{v_o}{r_{ds2}} + sC_2(v_o - v_{in}) + g_{m1}v_{in}=0
\Rightarrow
A_v(s)=\frac{sC_2-g_{m1}}{\tfrac{1}{r_{ds2}}+sC_2}.
\]
At low frequencies:
\[
\boxed{A_v\approx -g_{m1}r_{ds2}}.
\]
At $v_{in}$:
\[
i_{in}=sC_1v_{in}+sC_2(v_{in}-v_o)
=s\!\big[C_1+C_2(1-A_v)\big]\!v_{in},
\]
so
\[
\boxed{C_{\text{in}}=C_1+C_2(1-A_v)\approx C_1+C_2(1+|A_v|)}.
\]

\subsection*{Calculation \& Result}
\[
C_1=0.1~\mathrm{pF}, \;
C_2=0.05~\mathrm{pF}, \;
I_D=250~\mu\mathrm{A}, \;
V_{\text{eff1}}=0.859~\mathrm{V}, \;
\lambda_2=0.1~\mathrm{V^{-1}}.
\]
\[
g_{m1}=\frac{2I_D}{V_{\text{eff1}}}=0.582~\mathrm{mS}, \quad
r_{ds2}=\frac{1}{\lambda_2 I_D}=40~\mathrm{k}\Omega,
\]
\[
A_v=-0.582~\mathrm{mS}\times40~\mathrm{k}\Omega=-23.3, \quad
C_{\text{in}}=0.1+0.05(1+23.3)=\boxed{1.31~\mathrm{pF}}.
\]
